async function rateLimit(key: string) {
  const window = 60_000; // 1 min
  const limit = 1_000; // req/min
  const count = await redis
    .incr(`rl:${key}`);
  if (count === 1)
    await redis.pexpire(key, window);
  return count <= limit;
